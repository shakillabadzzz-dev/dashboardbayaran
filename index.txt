        // --- AFFILIATE DASHBOARD LOGIC ---
        function initAffiliateDashboard(userData) {
            document.getElementById('affEmailDisplay').innerText = userData.email;
            document.getElementById('affIdDisplay').innerText = userData.affiliateId;
            
            // Generate Mock Affiliate Link
            const baseUrl = window.location.origin + window.location.pathname;
            document.getElementById('affLink').value = `${baseUrl}?ref=${userData.affiliateId}`;

            // Listen to Wallet Data

                if (snapshot.exists()) {
                    const w = snapshot.val();
                    document.getElementById('statBalance').innerText = parseFloat(w.balance).toFixed(2);
                    document.getElementById('statSales').innerText = parseFloat(w.totalSales).toFixed(2);
                    document.getElementById('statComm').innerText = parseFloat(w.totalCommission).toFixed(2);
                    document.getElementById('statPending').innerText = parseFloat(w.pendingCommission).toFixed(2);
                }
            });

            // Listen to Withdrawal History
            const q = query(ref(db, 'withdrawals'), orderByChild('uid'), equalTo(currentUser.uid));
            onValue(q, (snapshot) => {
                const tbody = document.getElementById('historyTableBody');
                tbody.innerHTML = "";
                if (snapshot.exists()) {
                    // Sort descending by putting in array first
                    let dataArr = [];
                    snapshot.forEach(child => { dataArr.push(child.val()); });
                    dataArr.reverse().forEach(req => {
                        let statusColor = req.status === 'pending' ? 'text-orange-500' : (req.status === 'approved' ? 'text-green-500' : 'text-red-500');
                        let statusText = translations[currentLang][req.status] || req.status;
                        let dateStr = new Date(req.timestamp).toLocaleDateString();
                        
                        tbody.innerHTML += `
                            <tr class="border-b border-brand-50">
                                <td class="py-3">${dateStr}</td>
                                <td class="py-3 font-bold">RM ${parseFloat(req.amount).toFixed(2)}</td>
                                <td class="py-3 font-bold ${statusColor}">${statusText}</td>
                            </tr>
                        `;
                    });
                } else {
                    tbody.innerHTML = `<tr><td colspan="3" class="text-center py-4 text-gray-400">Tiada rekod</td></tr>`;
                }
            });

        // --- ADMIN DASHBOARD LOGIC ---
        function initAdminDashboard() {
            // Listen for ALL pending withdrawals
            onValue(ref(db, 'withdrawals'), (snapshot) => {
                const tbody = document.getElementById('adminWithdrawalsBody');
                tbody.innerHTML = "";
                
                if (snapshot.exists()) {
                    let hasPending = false;
                    snapshot.forEach(childSnap => {
                        const req = childSnap.val();
                        if (req.status === 'pending') {
                            hasPending = true;
                            let dateStr = new Date(req.timestamp).toLocaleDateString();
                            tbody.innerHTML += `
                                <tr class="border-b border-brand-50">
                                    <td class="py-3 text-brand-dark font-semibold">${req.email}</td>
                                    <td class="py-3 font-bold">RM ${parseFloat(req.amount).toFixed(2)}</td>
                                    <td class="py-3 text-xs">${dateStr}</td>
                                    <td class="py-3 flex gap-2">
                                        <button onclick="window.approveWithdrawal('${req.id}', '${req.uid}', ${req.amount})" class="bg-green-500 text-white px-3 py-1 rounded shadow hover:bg-green-600 text-xs font-bold"><i class="fa-solid fa-check"></i> Lulus</button>
                                        <button onclick="window.rejectWithdrawal('${req.id}')" class="bg-red-500 text-white px-3 py-1 rounded shadow hover:bg-red-600 text-xs font-bold"><i class="fa-solid fa-xmark"></i> Tolak</button>
                                    </td>
                                </tr>
                            `;
                        }
                    });
                    
                    if(!hasPending) tbody.innerHTML = `<tr><td colspan="4" class="text-center py-4 text-gray-400">Tiada permohonan baru.</td></tr>`;
                } else {
                    tbody.innerHTML = `<tr><td colspan="4" class="text-center py-4 text-gray-400">Tiada data.</td></tr>`;
                }
            });
        }

        // Global functions for Admin inline buttons
        window.approveWithdrawal = async (reqId, uid, amount) => {
            if(!confirm(`Luluskan pengeluaran RM${amount}? Baki wallet affiliate akan ditolak.`)) return;
            
            // 1. Get user wallet
            const walletRef = ref(db, 'wallets/' + uid);
            const wSnap = await get(walletRef);
            if (wSnap.exists()) {
                const currentBalance = wSnap.val().balance;
                if (currentBalance >= amount) {
                    // 2. Deduct Balance
                    await update(walletRef, { balance: currentBalance - amount });
                    // 3. Update Request Status
                    await update(ref(db, 'withdrawals/' + reqId), { status: 'approved' });
                    // 4. Record Transaction
                    await push(ref(db, 'transactions/' + uid), {
                        type: 'withdrawal',
